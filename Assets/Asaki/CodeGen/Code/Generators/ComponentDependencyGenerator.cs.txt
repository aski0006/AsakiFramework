using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace Asaki.CodeGen.Generators
{
    [Generator]
    public class ComponentDependencyGenerator : ISourceGenerator
    {
        public void Initialize(GeneratorInitializationContext context)
        {
            context.RegisterForSyntaxNotifications(() => new SyntaxReceiver());
        }

        public void Execute(GeneratorExecutionContext context)
        {
            if (!(context.SyntaxReceiver is SyntaxReceiver receiver)) return;

            var compilation = context.Compilation;
            var attributeSymbol = compilation.GetTypeByMetadataName("Asaki.Core.Architecture.Entities.ComponentDependencyAttribute");
            var interfaceSymbol = compilation.GetTypeByMetadataName("Asaki.Core.Architecture.Entities.IGeneratedDependencyInjector");

            if (attributeSymbol == null) return;

            // 按类分组处理字段
            var classGroups = receiver.CandidateFields
                .GroupBy(f => f.Parent.Parent as ClassDeclarationSyntax)
                .Where(g => g.Key != null);

            foreach (var group in classGroups)
            {
                var classDeclaration = group.Key;
                var model = compilation.GetSemanticModel(classDeclaration.SyntaxTree);
                var classSymbol = model.GetDeclaredSymbol(classDeclaration);

                if (classSymbol == null) continue;

                // 检查 partial 关键字
                if (!classDeclaration.Modifiers.Any(m => m.IsKind(SyntaxKind.PartialKeyword)))
                {
                    // 实际项目中可以输出 Diagnostic 报错，这里仅跳过
                    continue;
                }

                string source = GenerateClass(classSymbol, group.ToList(), model, attributeSymbol);
                string filename = $"{classSymbol.Name}_DependencyInjection.g.cs";
                context.AddSource(filename, source);
            }
        }

        private string GenerateClass(INamedTypeSymbol classSymbol, List<FieldDeclarationSyntax> fields, SemanticModel model, INamedTypeSymbol attrSymbol)
        {
            var sb = new StringBuilder();

            // 获取命名空间
            string namespaceName = classSymbol.ContainingNamespace.ToDisplayString();

            // 生成文件头
            sb.AppendLine("// <auto-generated/>");
            sb.AppendLine("using Asaki.Core.Architecture.Entities;");
            sb.AppendLine("using System;");
            sb.AppendLine();

            if (!string.IsNullOrEmpty(namespaceName))
            {
                sb.AppendLine($"namespace {namespaceName}");
                sb.AppendLine("{");
            }

            // 生成类定义 (partial) 并显式实现接口
            sb.AppendLine($"    partial class {classSymbol.Name} : IGeneratedDependencyInjector");
            sb.AppendLine("    {");
            sb.AppendLine("        void IGeneratedDependencyInjector.__Generated_InjectDependencies()");
            sb.AppendLine("        {");

            // 实体判空检查
            sb.AppendLine("            if (this.Entity == null) return;");
            sb.AppendLine();

            foreach (var fieldSyntax in fields)
            {
                foreach (var variable in fieldSyntax.Declaration.Variables)
                {
                    var fieldSymbol = model.GetDeclaredSymbol(variable) as IFieldSymbol;
                    if (fieldSymbol == null) continue;

                    // 检查特性
                    var attrData = fieldSymbol.GetAttributes().FirstOrDefault(ad => SymbolEqualityComparer.Default.Equals(ad.AttributeClass, attrSymbol));
                    if (attrData == null) continue;

                    // 获取字段类型
                    string typeName = fieldSymbol.Type.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);
                    string fieldName = fieldSymbol.Name;

                    // 获取 Required 属性值 (默认为 true)
                    bool isRequired = true;
                    if (attrData.NamedArguments.Length > 0)
                    {
                        foreach (var namedArg in attrData.NamedArguments)
                        {
                            if (namedArg.Key == "Required" && namedArg.Value.Value is bool val)
                            {
                                isRequired = val;
                                break;
                            }
                        }
                    }
                    else if (attrData.ConstructorArguments.Length > 0)
                    {
                         if(attrData.ConstructorArguments[0].Value is bool val) isRequired = val;
                    }

                    // 生成注入代码
                    // 这里的关键是：因为是 partial class，我们可以直接访问 this.privateField
                    sb.AppendLine($"            this.{fieldName} = this.GetComponent<{typeName}>();");

                    if (isRequired)
                    {
                        sb.AppendLine($"            if (this.{fieldName} == null)");
                        sb.AppendLine($"                throw new InvalidOperationException($\"[Asaki] Required component '{typeName}' not found on Entity {{this.Entity.Id}} for '{classSymbol.Name}.{fieldName}'\");");
                    }
                }
            }

            sb.AppendLine("        }");
            sb.AppendLine("    }");

            if (!string.IsNullOrEmpty(namespaceName))
            {
                sb.AppendLine("}");
            }

            return sb.ToString();
        }

        class SyntaxReceiver : ISyntaxReceiver
        {
            public List<FieldDeclarationSyntax> CandidateFields { get; } = new List<FieldDeclarationSyntax>();

            public void OnVisitSyntaxNode(SyntaxNode syntaxNode)
            {
                // 寻找带有特性的字段声明
                if (syntaxNode is FieldDeclarationSyntax fieldDeclaration && fieldDeclaration.AttributeLists.Count > 0)
                {
                    // 简单的名称检查过滤，后续在 Execute 中做精确的 Symbol 检查
                    if (fieldDeclaration.AttributeLists.Any(al => al.Attributes.Any(a => a.Name.ToString().Contains("ComponentDependency"))))
                    {
                        CandidateFields.Add(fieldDeclaration);
                    }
                }
            }
        }
    }
}
