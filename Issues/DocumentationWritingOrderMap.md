```
    │

    │
    ├── 依赖: Asaki.Core
    │
    ▼

    │
    ├── 依赖: Asaki.Core, Asaki.Generated

    ├── 依赖: Unity.Addressables, Unity.ResourceManager
    └── 依赖: Unity.TextMeshPro
    │
    ▼
Asaki.Editor (编辑器层)
    │
    ├── 依赖: Asaki.Core, Asaki.Unity
    ├── 依赖: UniTask, UniTask.Editor
    └── 依赖: Unity.TextMeshPro, UnityEditor.UI
    │
    ▼
Asaki.Plugins (插件扩展层)
    │
    ├── 依赖: Asaki.Core, Asaki.Unity, Asaki.Editor, Asaki.Generated
    └── 依赖: UniTask, Unity.TextMeshPro
```
