## 1. 元件化设计理念

### 1.1 核心原则

```
元件化思维:
├── 独立性 - 每个元件可独立使用
├── 可组合性 - 多个元件可组装成复杂单位
├── 可复用性 - 同一元件可用于不同场景
├── 可替换性 - 方便迭代和变体生成
└── 自动化友好 - 纯色背景便于脚本处理
```

### 1.2 背景处理规范

**抠图背景色**: `#00FF00` (纯绿色)

```
背景要求:
├── 纯色填充 - 无渐变、无纹理
├── 均匀分布 - 背景色值完全一致
├── 无阴影 - 元件周围无投影
├── 无光晕 - 边缘干净锐利
└── 无杂色 - 避免绿色出现在元件本身
```

**备选背景色**:
| 背景色 | HEX值 | 适用场景 |
|--------|-------|----------|
| 纯绿 | #00FF00 | 默认选择，与废土色调对比强 |
| 纯蓝 | #0000FF | 元件含大量绿色时使用 |
| 纯品红 | #FF00FF | 特殊情况备用 |
| 纯青 | #00FFFF | 另一备选方案 |

## 4. 防御塔元件详细设计

### 4.1 元件层级结构

```
防御塔 = 基座层 + 炮塔层 + 特效层

基座层 (Base Layer):
├── 静态元件 - 不参与旋转
├── 尺寸: 96x96 或 128x128
└── 锚点: 底部中心

炮塔层 (Turret Layer):
├── 动态元件 - 可360度旋转
├── 旋转帧: 16帧 (每帧22.5度)
├── 尺寸: 64x64 或 96x96
└── 锚点: 中心点

特效层 (Effect Layer):
├── 叠加元件 - 开火/充能效果
├── 动画帧: 4-8帧
└── 与炮塔同步旋转
```

```python
#!/usr/bin/env python3
"""
《撤离前线》元件化美术资产 - 自动抠图脚本
功能: 自动移除纯色背景(#00FF00)，生成透明PNG
依赖: pip install Pillow numpy
"""

import os
import sys
from pathlib import Path
from PIL import Image
import numpy as np
from typing import Tuple, List, Optional
import json
import argparse
from concurrent.futures import ThreadPoolExecutor, as_completed
import logging

# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger(__name__)


class BackgroundRemover:
    """背景移除处理器"""
    
    # 默认背景色
    DEFAULT_BG_COLOR = (0, 255, 0)  # #00FF00
    
    # 备选背景色
    ALTERNATE_BG_COLORS = [
        (0, 0, 255),      # #0000FF 纯蓝
        (255, 0, 255),    # #FF00FF 纯品红
        (0, 255, 255),    # #00FFFF 纯青
    ]
    
    def __init__(
        self,
        bg_color: Tuple[int, int, int] = None,
        tolerance: int = 30,
        edge_smoothing: bool = True,
        feather_pixels: int = 2
    ):
        """
        初始化背景移除器
        
        Args:
            bg_color: 背景色RGB值，默认#00FF00
            tolerance: 颜色容差(0-255)
            edge_smoothing: 是否启用边缘平滑
            feather_pixels: 边缘羽化像素数
        """
        self.bg_color = bg_color or self.DEFAULT_BG_COLOR
        self.tolerance = tolerance
        self.edge_smoothing = edge_smoothing
        self.feather_pixels = feather_pixels
        
    def detect_background_color(self, image: Image.Image) -> Optional[Tuple[int, int, int]]:
        """
        自动检测背景色
        
        检测图像四角的颜色，判断是否为纯色背景
        """
        img_array = np.array(image)
        height, width = img_array.shape[:2]
        
        # 采样四角区域
        corner_size = min(width, height) // 10
        corners = [
            img_array[0:corner_size, 0:corner_size],           # 左上
            img_array[0:corner_size, -corner_size:],           # 右上
            img_array[-corner_size:, 0:corner_size],           # 左下
            img_array[-corner_size:, -corner_size:],           # 右下
        ]
        
        # 检测每个角的主要颜色
        detected_colors = []
        for corner in corners:
            # 计算该区域最常见的颜色
            if len(corner.shape) == 3:  # RGB或RGBA
                pixels = corner.reshape(-1, corner.shape[-1])
                # 取前3个通道(RGB)
                if corner.shape[-1] >= 3:
                    pixels = pixels[:, :3]
            else:
                pixels = corner.reshape(-1, 1)
            
            # 计算平均颜色
            avg_color = tuple(np.mean(pixels, axis=0).astype(int))
            detected_colors.append(avg_color)
        
        # 检查四角颜色是否一致
        if len(set(detected_colors)) == 1:
            return detected_colors[0]
        
        # 检查是否匹配预设背景色
        for color in [self.DEFAULT_BG_COLOR] + self.ALTERNATE_BG_COLORS:
            matches = sum(
                1 for dc in detected_colors 
                if self._color_distance(dc, color) < self.tolerance
            )
            if matches >= 3:  # 至少3个角匹配
                return color
        
        return None
    
    def _color_distance(
        self, 
        color1: Tuple[int, int, int], 
        color2: Tuple[int, int, int]
    ) -> float:
        """计算两个颜色之间的欧氏距离"""
        return np.sqrt(sum((a - b) ** 2 for a, b in zip(color1, color2)))
    
    def remove_background(
        self, 
        image: Image.Image,
        auto_detect: bool = True
    ) -> Image.Image:
        """
        移除背景色，生成透明图像
        
        Args:
            image: 输入PIL图像
            auto_detect: 是否自动检测背景色
            
        Returns:
            带透明通道的PIL图像
        """
        # 确保图像为RGB模式
        if image.mode == 'RGBA':
            # 如果已有透明通道，先转为RGB
            background = Image.new('RGB', image.size, (255, 255, 255))
            background.paste(image, mask=image.split()[3])
            image = background
        elif image.mode != 'RGB':
            image = image.convert('RGB')
        
        # 自动检测或使用指定背景色
        if auto_detect:
            detected = self.detect_background_color(image)
            bg_color = detected if detected else self.bg_color
        else:
            bg_color = self.bg_color
        
        logger.debug(f"使用背景色: RGB{bg_color}")
        
        # 转换为numpy数组
        img_array = np.array(image)
        height, width = img_array.shape[:2]
        
        # 创建透明度掩码
        alpha_mask = np.ones((height, width), dtype=np.uint8) * 255
        
        # 计算每个像素与背景色的距离
        for y in range(height):
            for x in range(width):
                pixel = tuple(img_array[y, x])
                distance = self._color_distance(pixel, bg_color)
                
                if distance < self.tolerance:
                    # 完全透明
                    alpha_mask[y, x] = 0
                elif distance < self.tolerance * 2 and self.edge_smoothing:
                    # 边缘羽化
                    alpha = int(255 * (distance - self.tolerance) / self.tolerance)
                    alpha_mask[y, x] = alpha
        
        # 应用形态学操作平滑边缘
        if self.edge_smoothing:
            alpha_mask = self._smooth_edges(alpha_mask)
        
        # 创建RGBA图像
        rgba = np.dstack([img_array, alpha_mask])
        
        return Image.fromarray(rgba, 'RGBA')
    
    def _smooth_edges(self, alpha_mask: np.ndarray) -> np.ndarray:
        """使用形态学操作平滑边缘"""
        from scipy import ndimage
        
        # 轻微膨胀后腐蚀（闭运算）
        kernel_size = self.feather_pixels * 2 + 1
        kernel = np.ones((kernel_size, kernel_size))
        
        # 闭运算
        alpha_mask = ndimage.binary_closing(alpha_mask > 0, structure=kernel).astype(np.uint8) * 255
        
        # 高斯模糊羽化边缘
        alpha_mask = ndimage.gaussian_filter(alpha_mask.astype(float), sigma=self.feather_pixels)
        alpha_mask = np.clip(alpha_mask, 0, 255).astype(np.uint8)
        
        return alpha_mask
    
    def process_image(
        self,
        input_path: str,
        output_path: str,
        auto_detect: bool = True,
        trim: bool = True
    ) -> dict:
        """
        处理单个图像文件
        
        Args:
            input_path: 输入文件路径
            output_path: 输出文件路径
            auto_detect: 是否自动检测背景色
            trim: 是否裁剪透明边缘
            
        Returns:
            处理结果信息
        """
        try:
            # 打开图像
            image = Image.open(input_path)
            original_size = image.size
            
            # 移除背景
            result = self.remove_background(image, auto_detect)
            
            # 裁剪透明边缘
            if trim:
                bbox = result.getbbox()
                if bbox:
                    result = result.crop(bbox)
            
            # 确保输出目录存在
            os.makedirs(os.path.dirname(output_path), exist_ok=True)
            
            # 保存结果
            result.save(output_path, 'PNG')
            
            return {
                'input': input_path,
                'output': output_path,
                'original_size': original_size,
                'result_size': result.size,
                'success': True,
                'error': None
            }
            
        except Exception as e:
            logger.error(f"处理失败: {input_path}, 错误: {str(e)}")
            return {
                'input': input_path,
                'output': output_path,
                'success': False,
                'error': str(e)
            }


class BatchProcessor:
    """批量处理器"""
    
    def __init__(
        self,
        remover: BackgroundRemover,
        input_dir: str,
        output_dir: str,
        recursive: bool = True
    ):
        """
        初始化批量处理器
        
        Args:
            remover: 背景移除器实例
            input_dir: 输入目录
            output_dir: 输出目录
            recursive: 是否递归处理子目录
        """
        self.remover = remover
        self.input_dir = Path(input_dir)
        self.output_dir = Path(output_dir)
        self.recursive = recursive
        
        # 支持的图像格式
        self.supported_formats = {'.png', '.jpg', '.jpeg', '.bmp', '.tiff', '.webp'}
    
    def get_image_files(self) -> List[Path]:
        """获取所有待处理的图像文件"""
        if self.recursive:
            return [
                f for f in self.input_dir.rglob('*')
                if f.suffix.lower() in self.supported_formats
            ]
        else:
            return [
                f for f in self.input_dir.glob('*')
                if f.suffix.lower() in self.supported_formats
            ]
    
    def process_all(
        self,
        max_workers: int = 4,
        progress_callback: callable = None
    ) -> dict:
        """
        批量处理所有图像
        
        Args:
            max_workers: 最大并发数
            progress_callback: 进度回调函数
            
        Returns:
            处理统计信息
        """
        files = self.get_image_files()
        total = len(files)
        
        logger.info(f"找到 {total} 个图像文件待处理")
        
        results = {
            'total': total,
            'success': 0,
            'failed': 0,
            'details': []
        }
        
        def process_file(file_path: Path) -> dict:
            # 计算相对路径
            rel_path = file_path.relative_to(self.input_dir)
            output_path = self.output_dir / rel_path.with_suffix('.png')
            
            return self.remover.process_image(
                str(file_path),
                str(output_path)
            )
        
        # 使用线程池并行处理
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            futures = {executor.submit(process_file, f): f for f in files}
            
            for i, future in enumerate(as_completed(futures), 1):
                result = future.result()
                results['details'].append(result)
                
                if result['success']:
                    results['success'] += 1
                else:
                    results['failed'] += 1
                
                # 进度回调
                if progress_callback:
                    progress_callback(i, total, result)
                
                logger.info(f"进度: {i}/{total} - 成功: {results['success']}, 失败: {results['failed']}")
        
        return results

    """主函数"""
    parser = argparse.ArgumentParser(

        action='store_true',
        help='禁用边缘平滑'
    )
    parser.add_argument(
        '-w', '--workers',
        type=int,
        default=4,
        help='并发处理线程数（默认: 4）'
    )
    parser.add_argument(
        '-r', '--recursive',
        action='store_true',
        help='递归处理子目录'
    )
    parser.add_argument(
        '--report',
        help='处理报告输出路径（JSON格式）'
    )
    
    args = parser.parse_args()
    
    # 解析背景色
    bg_color = tuple(int(args.bg_color[i:i+2], 16) for i in (0, 2, 4))
    
    # 创建背景移除器
    remover = BackgroundRemover(
        bg_color=bg_color,
        tolerance=args.tolerance,
        edge_smoothing=not args.no_smooth
    )
    
    input_path = Path(args.input)
    
    # 判断是文件还是目录
    if input_path.is_file():
        # 处理单个文件
        output_path = args.output or str(input_path.with_name(
            input_path.stem + '_transparent' + input_path.suffix
        ))
        
        result = remover.process_image(
            str(input_path),
            output_path,
            auto_detect=not args.no_auto_detect,
            trim=not args.no_trim
        )
        
        if result['success']:
            logger.info(f"处理成功: {output_path}")
        else:
            logger.error(f"处理失败: {result['error']}")
            sys.exit(1)
            
    else:
        # 批量处理目录
        output_dir = args.output or str(input_path) + '_transparent'
        
        processor = BatchProcessor(
            remover=remover,
            input_dir=str(input_path),
            output_dir=output_dir,
            recursive=args.recursive
        )
        
        results = processor.process_all(max_workers=args.workers)
        
        # 输出统计
        logger.info("=" * 50)
        logger.info(f"处理完成!")
        logger.info(f"总计: {results['total']}")
        logger.info(f"成功: {results['success']}")
        logger.info(f"失败: {results['failed']}")
        
        # 保存报告
        if args.report:
            with open(args.report, 'w', encoding='utf-8') as f:
                json.dump(results, f, ensure_ascii=False, indent=2)
            logger.info(f"处理报告已保存: {args.report}")


if __name__ == '__main__':
    main()
```

```csharp
        public static void GenerateAtlas(string componentPath, AtlasConfig config)

            if (!Directory.Exists(sourcePath))
            {
                Debug.LogWarning($"[AtlasGenerator] 源目录不存在: {sourcePath}");
                return;
            }
            
            // 创建图集
            var atlas = new SpriteAtlas();
            
            // 配置打包设置
            var packingSettings = new SpriteAtlasPackingSettings
            {
                blockOffset = 1,
                enableRotation = config.AllowRotation,
                enableTightPacking = config.TightPacking,
                padding = config.Padding

            var textureSettings = new SpriteAtlasTextureSettings
            {
                readable = false,
                generateMipMaps = false,
                sRGB = true,
                filterMode = FilterMode.Point

            {

                format = TextureImporterFormat.RGBA32,

                format = TextureImporterFormat.ASTC_6x6,
                overridden = true

            atlas.SetPlatformSettings(mobileSettings);

            atlas.SetPlatformSettings(mobileSettings);

            var guids = AssetDatabase.FindAssets("t:Texture2D", new[] { sourcePath });
            var objects = guids
                .Select(AssetDatabase.GUIDToAssetPath)
                .Select(AssetDatabase.LoadAssetAtPath<Object>)
                .Where(obj => obj != null)
                .ToArray();
            
            if (objects.Length == 0)
            {
                Debug.LogWarning($"[AtlasGenerator] 未找到Sprite: {sourcePath}");
                return;
            }
            
            // 添加到图集
            atlas.Add(objects);
            
            // 保存图集
            string outputPath = Path.Combine(ATLAS_OUTPUT_PATH, config.Name + ".spriteatlas");
            Directory.CreateDirectory(Path.GetDirectoryName(outputPath));
            AssetDatabase.CreateAsset(atlas, outputPath);
            
            Debug.Log($"[AtlasGenerator] 生成图集: {config.Name}, 包含 {objects.Length} 个Sprite");
        }
    }
```
